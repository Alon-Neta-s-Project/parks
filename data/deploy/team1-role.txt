BEGIN;

set local search_path = public, extensions;

create role team1_content login password 'YOUR-TEAM1-PASSWORD-HERE';

grant connect on database postgres to team1_content;
grant usage on schema public     to team1_content;
grant usage on schema extensions to team1_content;

-- ════════════════════════════════════════════════════════════════════
-- 1 · גרנטים ברמת עמודה — מה שלא ניתן, לא ניתן לכתוב
-- ════════════════════════════════════════════════════════════════════

-- ⚠️ **אין `delete`, בשום טבלה.** מסמך קהילתי שגוי נפסל בשינוי
-- `review_status` ל-`rejected` ולא בהיעלמות. שורה שנמחקה אינה מותירה
-- עקבות של מה שהיה בה.
grant select on knowledge_doc, knowledge_chunk to team1_content;

-- 🔴 **`reviewed_by`, `reviewed_at` ו-`submitted_by` אינם ברשימה,
-- בשום פעולה.** לא ב-insert ולא ב-update. זו התשובה לשאלת גיא על
-- `submitted_by`: העמודה נשארת `NULL`, לא בגלל מוסכמה אלא כי
-- ל-`team1_content` אין הרשאה לכתוב אליה כלל. ניסיון ימות בהרשאה.
--
-- לצוות 1 אין משתמש. `NULL` אומר "לא הוגש על ידי משתמש", וזה נכון.
-- ערך placeholder קבוע היה אומר "הוגש על ידי X" — שקר שקט, ובדיוק
-- ההצהרה־שאיש־לא־בדק שאנחנו סופרים. (`submitted_by` הוא `uuid` בלי
-- מפתח זר — 003 — ולכן אין כאן שאלת FK כלל.)
grant insert (
  id, title, doc_type, authority_tier, locale,
  scope_resort, scope_park, scope_experience,
  source_urls, source_kind, country,
  volatility, last_verified, last_seen, corroboration_count,
  product_family, audience, v1_priority, purchase_type,
  body, review_status
) on knowledge_doc to team1_content;
-- `created_at`/`updated_at` אינם ברשימה: ברירת המחדל של המסד קובעת
-- אותן. כותב שיכול לקבוע `created_at` יכול לתארך שורה אחורה.

-- `review_status` כן ברשימת ה-update: דלי ב' שהתחזק לדלי א' עובר
-- מ-`draft` ל-`pending_review`. התקרה על הערך נאכפת למטה, לא כאן.
grant update (
  title, source_urls, country,
  volatility, last_verified, last_seen, corroboration_count,
  product_family, audience, v1_priority, purchase_type,
  body, review_status, updated_at
) on knowledge_doc to team1_content;

grant insert (
  doc_id, chunk_index, content,
  authority_tier, locale, scope_park, scope_experience, review_status
) on knowledge_chunk to team1_content;

-- ⚠️ **`embedding` ו-`embedding_model` אינם ברשימה.** הווקטור נוצר
-- בצינור ה-embed הקיים. הטריגר מ-033 מאפס אותו כש-`content` משתנה,
-- כך שעדכון גורר embedding מחדש מעצמו.
grant update (
  content, review_status
) on knowledge_chunk to team1_content;

-- ════════════════════════════════════════════════════════════════════
-- 2 · RLS — שתי מדיניות, ובכוונה
-- ════════════════════════════════════════════════════════════════════

-- המתירה אומרת **שיש גישה**. היא אינה מנסה לתאר את הכלל.
create policy knowledge_doc_team1 on knowledge_doc
  for all to team1_content using (true) with check (true);
create policy knowledge_chunk_team1 on knowledge_chunk
  for all to team1_content using (true) with check (true);

-- 🔴 **והמגבילה אומרת מהי הגישה — וזו היחידה שמכילה את הכלל.**
--
-- ⚠️ `restrictive` ולא `permissive`, ובכוונה. מדיניות מתירה נוספת
-- שמישהו יכתוב בעתיד **בלי `to`** חלה על כל התפקידים, ומדיניות מתירה
-- מצטרפת ב-OR — כלומר היא הייתה מרחיבה את `team1_content` בלי שאיש
-- ישים לב. `restrictive` מצטרפת ב-AND. אי אפשר לעקוף אותה בהוספה,
-- רק במחיקה מפורשת.
--
-- הכלל נכתב כאן פעם אחת. המתירה למעלה אינה חוזרת עליו.
create policy knowledge_doc_team1_cap on knowledge_doc
  as restrictive for all to team1_content
  using      (authority_tier = 'T4' and source_kind = 'community'
              and review_status in ('draft','pending_review'))
  with check (authority_tier = 'T4' and source_kind = 'community'
              and review_status in ('draft','pending_review'));

create policy knowledge_chunk_team1_cap on knowledge_chunk
  as restrictive for all to team1_content
  using      (authority_tier = 'T4'
              and review_status in ('draft','pending_review'))
  with check (authority_tier = 'T4'
              and review_status in ('draft','pending_review'));

-- 🔴 **`review_status` נמצא בשתי הפסקאות, ולא רק ב-`with check`.**
-- מצא על ידי גיא, 21.09, ותוקן לפני הרצה.
--
-- ה-`id` אידמפוטנטי לפי הפוסט: פוסט שנערך מייצר את אותו `id`, ולכן
-- איסוף חוזר **מעדכן שורה קיימת**. אם בינתיים אדם אישר אותה, והמסדר
-- כותב `review_status = 'draft'` כרגיל — עם `using` שאינו מסנן, העדכון
-- **מצליח**, והמסמך נופל מ-`approved` בחזרה ל-`draft`.
--
-- ⚠️ **ואז הוא נעלם מהתשובות של טים.** `match_knowledge` (028) מסננת
-- `review_status = 'approved'` בשתי הטבלאות. כלומר תוכן שהיה חי מפסיק
-- להישלף — בלי שגיאה, בלי שורה שנמחקה, ובלי שאיש ידע. זו בדיוק
-- המחלקה שהמוצר בנוי נגדה.
--
-- עם `review_status` ב-`using`, שורה מאושרת פשוט אינה נראית לתפקיד.
-- ⚠️ **ולכן ה-UPDATE יחזיר 0 שורות, ולא ישגה** — והמסדר חייב לקרוא את
-- מספר השורות שהושפעו ולהסלים, ולא להניח הצלחה. אפס שורות כאן פירושו
-- "אדם כבר אישר את זה", וזה מצב שדורש הכרעה ולא כתיבה מחדש.

-- מה שארבע הפסקאות האלה עושות, במילים:
--   `using`      — התפקיד אינו רואה ואינו נוגע בשורות T1–T3, **ולא
--                  בשורות שכבר אושרו**.
--   `with check` — מה שהוא כותב הוא T4 קהילתי, ו**לעולם לא**
--                  `'approved'` — לא ב-insert ולא ב-update.
--
-- 🔴 **זו התשובה לשאלתו של גיא על `approved`, והיא שכבתית:**
--   1. `with check` במדיניות המגבילה — פוסל את הערך עצמו.
--   2. גרנט ברמת עמודה — `reviewed_by`/`reviewed_at` אינם ניתנים
--      לכתיבה, כך ש"מאושר" לא יכול להיראות מאושר גם אם היה עובר.
-- שתיהן עצמאיות. נפילת אחת אינה פותחת את השנייה.

COMMIT;

-- 🔴 עמודת "עובר" חייבת להיות ✅ בכל שורה.
select 'התפקיד קיים' as "הבדיקה",
       (select count(*)::text from pg_roles where rolname = 'team1_content') as "יצא",
       '1' as "צפוי"
union all
select 'ואינו superuser',
       (select count(*)::text from pg_roles where rolname = 'team1_content'
          and (rolsuper or rolcreatedb or rolcreaterole)), '0'
union all
select 'אין לו delete בשום טבלה',
       (select count(*)::text from information_schema.role_table_grants
         where grantee = 'team1_content' and privilege_type = 'DELETE'), '0'
union all
select 'אינו נוגע ביומן המיגרציות',
       (select count(*)::text from information_schema.role_table_grants
         where grantee = 'team1_content' and table_name = 'schema_migration'), '0'
union all
select 'אינו נוגע בטבלאות משתמשים',
       (select count(*)::text from information_schema.role_table_grants
         where grantee in ('team1_content')
           and table_name in ('profile','trip','trip_member','conversation','message','turn_log')), '0'
union all
select 'אינו נוגע ב-park',
       (select count(*)::text from information_schema.role_table_grants
         where grantee = 'team1_content' and table_name = 'park'), '0'
union all
-- 🔴 **`privilege_type` — והיעדרו הוא מה שהפיל את שתי השורות האלה
-- בהרצה של 21.09.** `information_schema.column_privileges` פורשת גם
-- גרנט ברמת טבלה לשורה לכל עמודה. `grant select on knowledge_doc`
-- מייצר שם SELECT על `submitted_by`, `reviewed_by` ו-`reviewed_at` —
-- שלוש שורות, וזה בדיוק מה שיצא. שתיים נוספות מ-`knowledge_chunk`
-- על הווקטור.
--
-- ⚠️ כלומר הבדיקה שנועדה להוכיח **שאין כתיבה** ספרה **קריאה**.
-- הגרנטים היו תקינים כל הזמן. אותה משפחה כמו 011: בדיקה שמסתכלת על
-- הדבר הלא נכון ונכשלת בביטחון מלא. שוחזר במסד מקומי לפני התיקון.
select 'אין לו כתיבה ל-submitted_by',
       (select count(*)::text from information_schema.column_privileges
         where grantee = 'team1_content'
           and privilege_type in ('INSERT','UPDATE')
           and column_name in ('submitted_by','reviewed_by','reviewed_at')), '0'
union all
select 'אין לו כתיבה לווקטור',
       (select count(*)::text from information_schema.column_privileges
         where grantee = 'team1_content'
           and privilege_type in ('INSERT','UPDATE')
           and column_name in ('embedding','embedding_model')), '0'
union all
select 'המדיניות המגבילה קיימת על שתי הטבלאות',
       (select count(*)::text from pg_policies
         where policyname in ('knowledge_doc_team1_cap','knowledge_chunk_team1_cap')
           and permissive = 'RESTRICTIVE'), '2';
