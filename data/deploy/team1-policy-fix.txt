-- team1-policy-fix — החלפת שתי המדיניות המגבילות בלבד
-- ────────────────────────────────────────────────────────────────────
-- 📍 להריץ ב: Supabase ← SQL Editor
-- שם השאילתה: team1-policy-fix — review_status ב-using (22.09)
--
-- ⚠️ **להריץ רק אם `team1-recheck` החזיר "לא" בשורה האחרונה**
-- ("ושורה מאושרת אינה נראית לתפקיד"). אם החזיר "כן" — המדיניות
-- המתוקנת כבר במסד, ואין מה לעשות.
--
-- ⛔ **ולא להריץ את `team1-role.txt` שוב** — הוא יוצר את התפקיד
-- ויישבר על "role already exists". הקובץ הזה נוגע בשתי המדיניות
-- בלבד.
--
-- ── למה ─────────────────────────────────────────────────────────────
-- 🔴 **מסמך מאושר שנאסף שוב היה נופל בחזרה ל-draft ונעלם מהתשובות.**
--
-- ה-`id` אידמפוטנטי לפי הפוסט, ולכן איסוף חוזר מעדכן שורה קיימת ולא
-- יוצר חדשה. אם אדם אישר אותה בינתיים והמסדר כותב `'draft'` כרגיל —
-- עם `using` שאינו מסנן `review_status`, העדכון **מצליח**.
--
-- ואז המסמך נעלם: `match_knowledge` (028) מסננת `review_status =
-- 'approved'`. תוכן שהיה חי מפסיק להישלף — בלי שגיאה, בלי מחיקה,
-- ובלי שאיש ידע. (מצא גיא, 21.09.)
--
-- ⚠️ **ואחרי התיקון, `UPDATE` על שורה מאושרת מחזיר 0 שורות ואינו
-- שוגה.** המסדר חייב לקרוא את מספר השורות שהושפעו ולהסלים: אפס כאן
-- פירושו "אדם כבר אישר את זה" — הכרעה, לא כתיבה מחדש. זו דרישה על
-- הקוד של המסדר, ולא משהו שהמסד יאכוף במקומו.

BEGIN;

set local search_path = public, extensions;

drop policy if exists knowledge_doc_team1_cap   on knowledge_doc;
drop policy if exists knowledge_chunk_team1_cap on knowledge_chunk;

create policy knowledge_doc_team1_cap on knowledge_doc
  as restrictive for all to team1_content
  using      (authority_tier = 'T4' and source_kind = 'community'
              and review_status in ('draft','pending_review'))
  with check (authority_tier = 'T4' and source_kind = 'community'
              and review_status in ('draft','pending_review'));

create policy knowledge_chunk_team1_cap on knowledge_chunk
  as restrictive for all to team1_content
  using      (authority_tier = 'T4'
              and review_status in ('draft','pending_review'))
  with check (authority_tier = 'T4'
              and review_status in ('draft','pending_review'));

COMMIT;

-- ── אימות ───────────────────────────────────────────────────────────
-- 🔴 עמודת "עובר" חייבת להיות ✅ בכל שורה.
select 'שתי המדיניות קיימות ומגבילות' as "הבדיקה",
       (select count(*)::text from pg_policies
         where policyname in ('knowledge_doc_team1_cap','knowledge_chunk_team1_cap')
           and permissive = 'RESTRICTIVE') as "יצא",
       '2' as "צפוי"
union all
select 'ושתיהן מסננות review_status ב-using',
       (select count(*)::text from pg_policies
         where policyname in ('knowledge_doc_team1_cap','knowledge_chunk_team1_cap')
           and qual like '%review_status%'), '2'
union all
select 'וגם ב-with check',
       (select count(*)::text from pg_policies
         where policyname in ('knowledge_doc_team1_cap','knowledge_chunk_team1_cap')
           and with_check like '%review_status%'), '2';
