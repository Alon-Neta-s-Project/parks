-- 📍 להריץ ב: Supabase ← SQL Editor
-- שם השאילתה: set-tester-key — מפתח הבודק
-- ⚠️ אפשר להריץ שוב בכל פעם שרוצים להחליף מפתח.
--
-- ⚠️ **המפתח כבר ממולא בקובץ, ואין מה להחליף.** הוא אינו סוד: הוא
--    נוסע לדפדפן בכל מקרה, ולכן אין טעם שתבחרי אותו ותשמרי אותו בצד.
--    אם בכל זאת רוצים להחליף — לשנות את הערך בשורה אחת למטה.
--
-- ── מה זה עושה ───────────────────────────────────────────────────
--
-- המפתח הוא מה שמבדיל בין נטע לבין מי שנתקל בכתובת. בלעדיו ההערות
-- נשמרות בדפדפן ואינן מגיעות למסד.
--
-- ⚠️ **והוא אינו סוד חזק, ואינו מתיימר להיות.** הוא יושב בדפדפן, ומי
-- שפותח את כלי המפתחים רואה אותו. מה שהוא מונע: כתיבת שורות בידי מי
-- שאינו אמור. הטבלה מחזיקה הערות של הבודקות, לא נתוני משתמשים.
--
-- ⚠️ **ואי אפשר לקרוא אותו בחזרה מהמסך.** `tester_key()` מוסרת
-- מ-public ומ-anon במיגרציה 046 — כלומר גם מי שמחזיק את המפתח
-- הציבורי אינו יכול לשאול מה הוא. לכן אם שכחת אותו, לא שולפים —
-- מריצים את הקובץ הזה שוב עם ערך חדש.

BEGIN;

set local search_path = public, extensions;

create or replace function public.tester_key() returns text
language sql immutable parallel safe
as $$ select 'tester-wicx8w4nm22n' $$;

-- ⚠️ ההרשאה מוסרת שוב: `create or replace` משחזרת את ברירת המחדל,
-- שהיא execute לכולם. בלי השורה הזו המפתח היה נקרא מהדפדפן.
revoke all on function public.tester_key() from public, anon, authenticated;

COMMIT;

-- ── אימות ────────────────────────────────────────────────────────
-- 📍 להריץ בנפרד, אחרי ה-COMMIT.
--
-- ⚠️ **הערך עצמו אינו מוחזר כאן, בכוונה.** מה שנבדק: שהוא השתנה,
-- שאינו הערך המקורי מהקובץ, ושאיש אינו יכול לקרוא אותו.

-- 🔴 **הבדיקה אינה מזכירה את טקסט המציין.** בגרסה הראשונה היא השוותה
-- מולו מילה במילה, ואז החלפת־הכל בעורך שינתה גם אותה — והיא דיווחה
-- "לא הוחלף" על מפתח שהוחלף. גלאי שמכיל את מה שהוא מחפש נשבר בדיוק
-- כשמשתמשים בו כמתוכנן. זו הפעם החמישית בסשן הזה.
select 'המפתח אינו מציין' as "מה נבדק",
       (case when public.tester_key() ilike '%your%' or public.tester_key() ilike '%here%'
             then 'עדיין מציין' else 'הוחלף ✅' end) as "בפועל",
       'הוחלף ✅' as "מצופה"
union all
select 'אורך המפתח',
       length(public.tester_key())::text, '8 ומעלה'
union all
select 'anon אינו יכול לקרוא אותו',
       (select count(*)::text from information_schema.routine_privileges
         where routine_name = 'tester_key'
           and grantee in ('anon','authenticated','PUBLIC')), '0';
