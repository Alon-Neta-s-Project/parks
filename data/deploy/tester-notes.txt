-- tester-notes — מה נכתב בסבב הבדיקה
-- ────────────────────────────────────────────────────────────────────
-- 📍 להריץ ב: Supabase ← SQL Editor
-- שם השאילתה: tester-notes (כלי חוזר — לא צריך מספר)
--
-- ⛔ **קורא בלבד.** אינו משנה דבר, ואפשר להריץ שוב ושוב.
-- ⚠️ דורש שמיגרציה 046 כבר רצה.
--
-- להריץ אחרי סבב בדיקה, ולהדביק לי את כל הטבלה.

select
  to_char(created_at at time zone 'Asia/Jerusalem', 'DD/MM HH24:MI') as "מתי",
  session_ref  as "סבב",
  turn_ref     as "תור",
  note         as "ההערה",
  coalesce(left(question, 120), '—') as "השאלה",
  coalesce(left(answer,   200), '—') as "מה טים ענה"
from public.tester_notes(200)
order by created_at;
